        def determinar_temporal_modalidad(modalidad):
            texto_local = texto

            # =========================================================
            # 1. MODALIDADES NEGATIVAS
            # =========================================================

            if modalidad == "NO ELABORACIÓN":
                if re.search(
                    r"\bno fue elaborad[oa]\b.{0,45}\bdentro del plazo\b|"
                    r"\bno elabor[oó]\b.{0,45}\bdentro del plazo\b",
                    texto_local,
                    flags=re.IGNORECASE,
                ):
                    return "INCUMPLIMIENTO DE PLAZO"
                return "SIN INFORMACIÓN TEMPORAL"

            if modalidad == "NO REMISIÓN":
                if re.search(
                    r"\bno fue remitid[oa]\b.{0,45}\bdentro del plazo\b|"
                    r"\bno remiti[oó]\b.{0,45}\bdentro del plazo\b|"
                    r"\bni remitid[oa]\b.{0,45}\bdentro del plazo\b",
                    texto_local,
                    flags=re.IGNORECASE,
                ):
                    return "INCUMPLIMIENTO DE PLAZO"
                return "SIN INFORMACIÓN TEMPORAL"

            # =========================================================
            # 2. PATRONES DE ACCIÓN
            # =========================================================

            patrones_accion = {
                "ELABORACIÓN": [
                    r"\bfue elaborado\b",
                    r"\bfue elaborada\b",
                    r"\bha sido elaborado\b",
                    r"\bha sido elaborada\b",
                    r"\belabor[oó]\b",
                    r"\belaborado\b",
                    r"\belaborada\b",
                ],
                "REMISIÓN": [
                    r"\bfue remitido\b",
                    r"\bfue remitida\b",
                    r"\bha sido remitido\b",
                    r"\bha sido remitida\b",
                    r"\bremiti[oó]\b",
                    r"\bremitido\b",
                    r"\bremitida\b",
                ],
            }

            # =========================================================
            # 3. LOCALIZAR TODAS LAS ACCIONES
            # =========================================================

            ocurrencias = []

            for tipo, patrones in patrones_accion.items():
                for patron in patrones:
                    for match in re.finditer(
                        patron,
                        texto_local,
                        flags=re.IGNORECASE,
                    ):
                        ocurrencias.append(
                            {
                                "tipo": tipo,
                                "inicio": match.start(),
                                "fin": match.end(),
                            }
                        )

            # Eliminar duplicados exactos producidos por patrones
            # generales que contienen otros patrones más específicos.
            unicas = {}

            for ocurrencia in ocurrencias:
                clave = (
                    ocurrencia["tipo"],
                    ocurrencia["inicio"],
                    ocurrencia["fin"],
                )
                unicas[clave] = ocurrencia

            ocurrencias = list(unicas.values())
            ocurrencias.sort(key=lambda x: x["inicio"])

            ocurrencias_modalidad = [
                o for o in ocurrencias
                if o["tipo"] == modalidad
            ]

            if not ocurrencias_modalidad:
                return "SIN INFORMACIÓN TEMPORAL"

            # =========================================================
            # 4. CLASIFICADOR DE TEMPORALIDAD DE UN SEGMENTO
            # =========================================================

            def clasificar_segmento(segmento):
                if re.search(
                    r"\bfuera del plazo\b|"
                    r"\bdespu[eé]s del plazo\b|"
                    r"\bposterior al plazo\b|"
                    r"\bextempor[aá]neamente\b|"
                    r"\btard[ií]amente\b",
                    segmento,
                    flags=re.IGNORECASE,
                ):
                    return "INCUMPLIMIENTO DE PLAZO"

                if re.search(
                    r"\bdentro del plazo\b|"
                    r"\boportunamente\b|"
                    r"\bde manera oportuna\b",
                    segmento,
                    flags=re.IGNORECASE,
                ):
                    return "CUMPLIMIENTO DE PLAZO"

                return None

            # =========================================================
            # 5. ANALIZAR TEMPORALIDAD INDIVIDUAL
            #
            # Cada acción se analiza solamente hasta la siguiente
            # acción. Esto evita que la temporalidad de REMISIÓN
            # contamine ELABORACIÓN, y viceversa.
            # =========================================================

            estados = []

            for ocurrencia in ocurrencias_modalidad:
                inicio = ocurrencia["inicio"]
                fin = ocurrencia["fin"]

                siguientes = [
                    o["inicio"]
                    for o in ocurrencias
                    if o["inicio"] > inicio
                ]

                limite = min(siguientes) if siguientes else len(texto_local)

                segmento = texto_local[fin:limite]

                # Una modalidad negativa posterior constituye una
                # frontera semántica. Su temporalidad no debe
                # atribuirse a la acción positiva anterior.
                segmento = re.split(
                    r"\bno fue remitid[oa]\b|"
                    r"\bno remiti[oó]\b|"
                    r"\bno fue elaborad[oa]\b|"
                    r"\bno elabor[oó]\b",
                    segmento,
                    maxsplit=1,
                    flags=re.IGNORECASE,
                )[0]

                # Una conjunción adversativa separa la temporalidad
                # de la acción anterior de una nueva proposición.
                segmento = re.split(
                    r"\b(?:pero|sin embargo|aunque)\b",
                    segmento,
                    maxsplit=1,
                    flags=re.IGNORECASE,
                )[0]

                estado = clasificar_segmento(segmento)

                estados.append(
                    {
                        "inicio": inicio,
                        "fin": fin,
                        "estado": estado,
                    }
                )

            # =========================================================
            # 6. SI EXISTE TEMPORALIDAD INDIVIDUAL, TIENE PRIORIDAD
            # =========================================================

            estados_conocidos = [
                estado["estado"]
                for estado in estados
                if estado["estado"] is not None
            ]

            if estados_conocidos:
                # Si esta modalidad tiene una temporalidad explícita,
                # devolvemos esa temporalidad.
                for estado in estados:
                    if estado["estado"] is not None:
                        return estado["estado"]

            # =========================================================
            # 7. TEMPORALIDAD COMPARTIDA EN COORDINACIONES
            #
            # Ejemplo:
            # "fue elaborado y luego remitido dentro del plazo"
            #
            # La primera acción no tiene temporalidad propia en su
            # segmento; la segunda sí. En una coordinación válida,
            # heredamos la temporalidad de la segunda acción.
            # =========================================================

            if len(ocurrencias) >= 2:

                for indice in range(len(ocurrencias) - 1):

                    primera = ocurrencias[indice]
                    segunda = ocurrencias[indice + 1]

                    if primera["tipo"] == segunda["tipo"]:
                        continue

                    texto_entre = texto_local[
                        primera["fin"]:segunda["inicio"]
                    ]

                    if not re.search(
                        r"\b(?:y|luego|posteriormente)\b",
                        texto_entre,
                        flags=re.IGNORECASE,
                    ):
                        continue

                    segmento_segunda = texto_local[
                        segunda["fin"]:
                    ]

                    estado_segunda = clasificar_segmento(
                        segmento_segunda
                    )

                    if estado_segunda is None:
                        continue

                    if (
                        primera["tipo"] == modalidad
                        or segunda["tipo"] == modalidad
                    ):
                        return estado_segunda

            # =========================================================
            # 8. SIN INFORMACIÓN TEMPORAL
            # =========================================================

            return "SIN INFORMACIÓN TEMPORAL"
